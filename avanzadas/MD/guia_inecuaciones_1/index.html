    \paragraph{Ejemplos resueltos.} Antes de comenzar con los ejercicios, revisemos un ejemplo de cada tipo de inecuación presente en este bloque. Lea con atención tanto la materia que se recuerda como cada paso del desarrollo: es exactamente el tipo de trabajo que se espera que usted realice por su cuenta.

    %------------------------------------------------------------------------ \bigskip \noindent\textbf{Ejemplo 1 (Inecuación cuadrática).} Encuentre el conjunto solución de [ 3x^{2}-5x-2>0 . ]

    \medskip \noindent\textit{Materia necesaria.} \begin{itemize} \item Una inecuación cuadrática es de la forma \(ax^{2}+bx+c \;\square\; 0\), con \(a\neq 0\) y \(\square\in\{<,\leq,>,\geq\}\). \item \textbf{Discriminante:} \(\Delta=b^{2}-4ac\). Las raíces de \(ax^{2}+bx+c=0\) son \(x=\dfrac{-b\pm\sqrt{\Delta}}{2a}\). \begin{itemize} \item Si \(\Delta>0\): hay dos raíces reales distintas \(x_1<x_2\) y \(ax^{2}+bx+c=a(x-x_1)(x-x_2)\). \item Si \(\Delta=0\): hay una raíz doble \(x_0\) y \(ax^{2}+bx+c=a(x-x_0)^{2}\), que nunca cambia de signo. \item Si \(\Delta<0\): no hay raíces reales y la expresión tiene \emph{siempre} el signo de \(a\). \end{itemize} \item \textbf{Regla de signos de un producto:} un producto es positivo si tiene una cantidad par de factores negativos, y negativo si tiene una cantidad impar. \item \textbf{Consecuencia útil:} cuando \(\Delta>0\), el trinomio tiene el signo de \(a\) \emph{fuera} de las raíces y el signo contrario \emph{entre} ellas. \end{itemize}

    \medskip \noindent\textit{Desarrollo.}

    \noindent\textbf{Paso 1: Calcular las raíces.} Aquí \(a=3\), \(b=-5\), \(c=-2\), luego [ \Delta=(-5)^{2}-4(3)(-2)=25+24=49>0, \qquad x=\frac{5\pm\sqrt{49}}{6}=\frac{5\pm 7}{6} \;\Longrightarrow\; x_1=-\frac{1}{3},\quad x_2=2 . ]

    \noindent\textbf{Paso 2: Factorizar.} [ 3x^{2}-5x-2=3\left(x+\tfrac{1}{3}\right)(x-2)=(3x+1)(x-2). ] (Comprobación: \((3x+1)(x-2)=3x^{2}-6x+x-2=3x^{2}-5x-2\). \checkmark)

    \noindent\textbf{Paso 3: Tabla de signos.} Los puntos críticos \(-\tfrac13\) y \(2\) dividen la recta real en tres intervalos. En cada uno, el signo de cada factor es constante: [ \begin{array}{c|ccc} & \left(-\infty,-\frac{1}{3}\right) & \left(-\frac{1}{3},2\right) & (2,+\infty)\ \hline 3x+1 & - & + & + \ x-2 & - & - & + \ \hline (3x+1)(x-2) & + & - & + \end{array} ]

    \noindent\textbf{Paso 4: Seleccionar.} Buscamos donde el producto es \emph{estrictamente} positivo, por lo que los puntos críticos (donde vale \(0\)) \emph{no} se incluyen: [ \boxed{\;S=\left(-\infty,-\tfrac{1}{3}\right)\cup(2,+\infty)\;} ]

    \noindent\textbf{Verificación.} Con \(x=0\notin S\): \(3(0)^{2}-5(0)-2=-2>0\) es falso. Con \(x=3\in S\): \(27-15-2=10>0\) es verdadero. \checkmark

    %------------------------------------------------------------------------ \bigskip \noindent\textbf{Ejemplo 2 (Inecuación racional).} Resuelva [ \frac{x^{3}+x}{x^{2}-4}\leq\frac{2x^{2}}{x^{2}-4}. ]

    \medskip \noindent\textit{Materia necesaria.} \begin{itemize} \item \textbf{No se multiplica por el denominador} si no se conoce su signo: multiplicar por una cantidad negativa invierte la desigualdad. Por eso se lleva todo a un solo lado y se compara con cero: \(\dfrac{P(x)}{Q(x)}\;\square\;0\). \item El cociente \(\dfrac{P}{Q}\) tiene el mismo signo que el producto \(P\cdot Q\), pero \textbf{los ceros de \(Q\) se excluyen siempre} (la expresión no está definida). \item Los ceros de \(P\) se incluyen solo si la desigualdad es no estricta (\(\leq\) o \(\geq\)). \item \textbf{Multiplicidad:} si un factor aparece como \((x-r)^{k}\), \begin{itemize} \item con \(k\) impar, la expresión \emph{cambia} de signo al pasar por \(r\); \item con \(k\) par, la expresión \emph{no cambia} de signo al pasar por \(r\) (pues \((x-r)^{k}\geq 0\) siempre). \end{itemize} \end{itemize}

    \medskip \noindent\textit{Desarrollo.}

    \noindent\textbf{Paso 1: Restricciones.} \(x^{2}-4\neq 0 \iff x\neq -2 \text{ y } x\neq 2\).

    \noindent\textbf{Paso 2: Llevar todo a un lado.} [ \frac{x^{3}+x}{x^{2}-4}-\frac{2x^{2}}{x^{2}-4}\leq 0 \iff \frac{x^{3}-2x^{2}+x}{x^{2}-4}\leq 0 . ]

    \noindent\textbf{Paso 3: Factorizar numerador y denominador.} [ x^{3}-2x^{2}+x=x\,(x^{2}-2x+1)=x\,(x-1)^{2}, \qquad x^{2}-4=(x-2)(x+2), ] así que la inecuación es equivalente a [ \frac{x\,(x-1)^{2}}{(x-2)(x+2)}\leq 0,\qquad x\neq\pm 2 . ]

    \noindent\textbf{Paso 4: Puntos críticos.} Numerador: \(x=0\) (multiplicidad \(1\)) y \(x=1\) (multiplicidad \(2\)). Denominador: \(x=-2\) y \(x=2\) (excluidos). Ordenados: \(-2<0<1<2\).

    \noindent\textbf{Paso 5: Tabla de signos.} [ \begin{array}{c|ccccc} & (-\infty,-2) & (-2,0) & (0,1) & (1,2) & (2,+\infty)\ \hline x & - & - & + & + & + \ (x-1)^{2} & + & + & + & + & + \ x+2 & - & + & + & + & + \ x-2 & - & - & - & - & + \ \hline \dfrac{x(x-1)^{2}}{(x-2)(x+2)} & - & + & - & - & + \end{array} ] Observe que en \(x=1\) la expresión \emph{no cambia} de signo (factor de multiplicidad par), tal como predice la teoría.

    \noindent\textbf{Paso 6: Seleccionar.} Queremos \(\leq 0\): \begin{itemize} \item intervalos negativos: \((-\infty,-2)\), \((0,1)\) y \((1,2)\); \item ceros del numerador (se incluyen por ser \(\leq\)): \(x=0\) y \(x=1\); \item \(x=\pm 2\) se excluyen por anular el denominador. \end{itemize} Uniendo \((0,1)\cup\{1\}\cup(1,2)\cup\{0\}=[0,2)\), se obtiene [ \boxed{\;S=(-\infty,-2)\cup[0,2)\;} ]

    \noindent\textbf{Verificación.} Con \(x=-3\): \(\dfrac{(-3)(16)}{(-5)(-1)}=-\dfrac{48}{5}\leq 0\) \checkmark. Con \(x=-1\notin S\): \(\dfrac{(-1)(4)}{(-3)(1)}=\dfrac{4}{3}\leq 0\) es falso \checkmark.

    \noindent\textbf{Error típico.} Si se ``cancela'' el denominador multiplicando por \(x^{2}-4\) sin analizar su signo, se obtiene \(x(x-1)^{2}\leq 0\), es decir \(x\leq 0\), lo cual es \emph{incorrecto} (por ejemplo, \(x=-1\) no es solución).

    %------------------------------------------------------------------------ \bigskip \noindent\textbf{Ejemplo 3 (Inecuación irracional).} Determine todos los \(x\in\R\) tales que [ \sqrt{x+7}>x+1 . ]

    \medskip \noindent\textit{Materia necesaria.} \begin{itemize} \item \textbf{Dominio:} \(\sqrt{A}\) está definida en \(\R\) solo si \(A\geq 0\), y en ese caso \(\sqrt{A}\geq 0\). \item \textbf{Elevar al cuadrado} conserva la desigualdad \emph{solo si ambos lados son no negativos}: si \(u,v\geq 0\), entonces \(u<v\iff u^{2}<v^{2}\). Si algún lado es negativo esto falla: \(-3<2\), pero \((-3)^{2}>2^{2}\). \item De lo anterior se obtienen las equivalencias estándar: [ \sqrt{A}>B\iff \underbrace{\big(A\geq 0\ \wedge\ B<0\big)}{\text{Caso 1}} \ \vee\ \underbrace{\big(B\geq 0\ \wedge\ A>B^{2}\big)}, ] [ \sqrt{A}0\ \wedge\ A<B^{2}. ] (Para }\(\geq\) y \(\leq\) se procede igual, cambiando las desigualdades estrictas correspondientes.) \end{itemize}

    \medskip \noindent\textit{Desarrollo.} Aquí \(A=x+7\) y \(B=x+1\).

    \noindent\textbf{Paso 1: Dominio.} \(x+7\geq 0\iff x\geq -7\).

    \noindent\textbf{Paso 2: Caso 1 (\(B<0\)).} Si \(x+1<0\), es decir \(x<-1\), el lado derecho es negativo y el izquierdo es \(\geq 0\), por lo que la desigualdad se cumple automáticamente (siempre que la raíz esté definida). Intersecando con el dominio: [ S_1=[-7,+\infty)\cap(-\infty,-1)=[-7,-1). ]

    \noindent\textbf{Paso 3: Caso 2 (\(B\geq 0\)).} Si \(x\geq -1\), ambos lados son no negativos y podemos elevar al cuadrado: [ x+7>(x+1)^{2} \iff x+7>x^{2}+2x+1 \iff x^{2}+x-6<0 \iff (x+3)(x-2)<0 . ] Por lo visto en el Ejemplo 1 (el trinomio tiene signo contrario al de \(a=1\) entre las raíces), esto ocurre para \(-3<x<2\). Intersecando con la condición del caso (la condición \(A\geq 0\) ya se cumple, pues \(A>B^{2}\geq 0\)): [ S_2=[-1,+\infty)\cap(-3,2)=[-1,2). ]

    \noindent\textbf{Paso 4: Unir los casos.} [ \boxed{\;S=S_1\cup S_2=[-7,-1)\cup[-1,2)=[-7,2)\;} ]

    \noindent\textbf{Verificación.} Con \(x=-7\): \(\sqrt{0}=0>-6\) \checkmark. Con \(x=2\) (excluido): \(\sqrt{9}=3>3\) es falso \checkmark. Con \(x=3\notin S\): \(\sqrt{10}\approx 3{,}16>4\) es falso \checkmark.

    \noindent\textbf{Error típico.} Elevar al cuadrado directamente sin separar casos entrega solo \((-3,2)\): se \emph{pierden} las soluciones de \([-7,-3]\), donde el lado derecho es negativo.
